import { test } from "vitest";
import fs from "node:fs";
import { readArrivalResponse, toObservations, anchorArrivals, parseRecptnDt } from "@/lib/arrivals";
import { judge } from "@/lib/verdict";
import { DEFAULT_SETTINGS } from "@/stores/settings-store";
test("demo", () => {
  for (const [f, sid] of [["onsu-7.json","1007"],["seongsu-2.json","1002"],["bonghwasan-6.json","1006"],["kkachisan.json","1005"]] as const) {
    const body = JSON.parse(fs.readFileSync(`src/lib/__fixtures__/${f}`, "utf8"));
    const { list } = readArrivalResponse(body);
    if (!list.length) continue;
    const now = parseRecptnDt(list[0].recptnDt) + 45_000; // 받는 시점 ≈ 생성 + 45초
    const obs = toObservations(list, sid, now);
    const { arrivals } = anchorArrivals({}, obs);
    for (const dir of ["up","down"] as const) {
      const a = arrivals.filter(x => x.direction === dir);
      const v = judge(a, now, { walkSec: 300 }, DEFAULT_SETTINGS);
      const nearest = a[0] ? `${a[0].message} / 계산 ${Math.round((a[0].arrivalAt-now)/1000)}초` : "-";
      const shown = v.kind === "walk" || v.kind === "run" ? `${v.kind} ${Math.round(v.main.remainingSec)}초 (${v.main.arrival.message}), 건너뛴 열차 ${v.missed.length}대` : v.kind;
      console.log(f, dir, "| 가장 가까운:", nearest, "| 카드:", shown);
    }
  }
});
